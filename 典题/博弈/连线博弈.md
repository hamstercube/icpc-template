#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
ll f[100100];
mt19937_64 rnd(time(0));

ll sg(ll x){
	if(x > 103) return sg((x-70)%34+70);
	if(f[x] != -1) return f[x];
	unordered_set<ll> s;
	for(ll i = 0; i <= x-2; i++){
		s.insert(sg(i)^sg(x-2-i));
	}
	ll i = 0;
	while(s.count(i)) i++;
	return f[x] = i; 
}
// 70-103

void solve(){
	cin >> n >> m;
	map<ll, ll> w, len;
	for(ll i = 0; i < m; i++){
		ll t1, t2;
		cin >> t1 >> t2;
		ll t = rnd();
		w[t1] ^= t, w[t2] ^= t;
	}
	ll ans = 0, p = 0, pre = -1;
	w[n] = 0;
	for(auto [x, y] : w){
		len[p] += x-pre-1;
		p ^= y, pre = x; 
	}
	for(auto [x, y] : len) ans ^= sg(y);
	if(ans) cout << "YES" << endl;
	else cout << "NO" << endl;
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	memset(f, -1, sizeof f);
//	for(ll i = 0; i <= 103; i++) sg(i);
	ll T;
	cin >> T;
	while(T--){
		solve();
	}
//	for(ll i = 0; i <= 1000; i++){
//		cout << i << ": " << sg(i) << endl;
//	}
	return 0;
}
