左右脑互搏（状态压缩）

```cpp
#include<bits/stdc++.h>
// #define endl '\n'
using namespace std;
typedef int ll;

ll i, j, n, m, a, b;
ll v[2000100];
ll f[2000100];

bool sg(ll x, ll sum){
	if(f[x] != -1) return f[x];
	for(ll i = 0, j = 1; i < 20; i++, j*=2){
		if((x&j) && (sum^v[i]) < v[i]){
			if(sg(x-j, sum^v[i]) == 0){
				return f[x] = 1;
			}
		}
	}
	return f[x] = 0;
}

void solve(){
	memset(f, -1, sizeof f);
	cin >> n;
	ll sum = 0;
	for(ll i = 0; i < n; i++){
		cin >> v[i];
		sum ^= v[i];
	}
	ll k = 1;
	while(n--) k*=2;
	k--;
	if(sg(k, sum)) cout << "Left" << endl;
	else cout << "Right" << endl;
}

signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    ll T = 1;
	cin >> T;
    while (T--) solve();
    return 0;
}
```
