求两人相遇且走的两人走的步数最大值最小

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;

ll ex_gcd(ll a, ll b, __int128 &x, __int128 &y){
	if(b == 0){
		x = 1, y = 0;
		return a;
	}
	ll res = ex_gcd(b ,a%b, y, x);
	y = y-a/b*x;
	return res;
}

void solve(){
    ll x, a, s;
    __int128 xt, yt;
    cin >> x >> a >> s;
    ll g = ex_gcd(a, s, xt, yt);
    if(x%g){  //绝对不能相遇
        cout << "No" << endl;
        return ;
    }
    array<__int128, 2> ans = {-1, -1};
    auto check = [&](array<__int128, 2> a){
//         cout << a[0] << " " << a[1] << endl;
        if(a[0] < 0 || a[1] < 0) return ;
        if(ans[0] == -1 || max(ans[0], ans[1]) > max(a[0], a[1])) ans = a;
    };
    xt *= x/g, yt *= x/g;
    __int128 k = (xt-yt)/((a+s)/g);
    __int128 k1 = xt-k*s/g, k2 = yt+k*a/g;
    check({k1, k2});
    check({k1+s/g, k2-a/g});
    check({k1-s/g, k2+a/g});
    if(ans[0] == -1){
        cout << "No" << endl;
        return ;
    }
    cout << "Yes" << endl;
    cout << (ll)ans[0] << " " << (ll)ans[1] << endl;
}

int main(){
	ll T;
    cin >> T;
    while(T--) solve();
	return 0;
}
```
