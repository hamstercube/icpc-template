#include <bits/stdc++.h>
#define int long long
using namespace std;

const int N = 1e7 + 10, M = 2010, mod = 1e9 + 7;
int n, m, k, idx = 0, flag = 0; // flag=1表示无解

/**
 * 扩展欧几里得算法：求解 ax + by = gcd(a,b) 的一组整数解 (x,y)
 */
int exgcd(int a, int b, int &x, int &y)
{
    if (b == 0) // 递归边界：b=0时，a*1 + 0*0 = a = gcd(a,0)
    {
        x = 1, y = 0;
        return a;
    }
    int x1, y1;
    int gcd = exgcd(b, a % b, x1, y1); // 递归求解子问题
    x = y1, y = x1 - a / b * y1;
    return gcd;
}

signed main()
{
    scanf("%lld", &n);
    int a1, m1;
    scanf("%lld %lld", &a1, &m1); // 第一个方程：x ≡ m1 (mod a1)

    // 合并n-1个方程，最终合并为 x ≡ m1 (mod a1) 的形式
    while (n-- > 1)
    {
        int a2, m2;
        scanf("%lld %lld", &a2, &m2); // 下一个方程：x ≡ m2 (mod a2)

        int x, y;
        // 扩展欧几里得求解 a1*x + a2*y = gcd(a1,a2)
        int d = exgcd(a1, a2, x, y);

        // 合并两个方程的前提：(m2 - m1) 必须能被 gcd(a1,a2) 整除
        // 推导：x = m1 + k1*a1，x = m2 + k2*a2 → a1*k1 - a2*k2 = m2 - m1
        // 该方程有解的充要条件是 gcd(a1,a2) | (m2 - m1)
        if ((m2 - m1) % d)
        {
            flag = 1;
            break;
        }

        // 扩大x，使得方程满足 a1*x + a2*y = m2 - m1
        x = x * (m2 - m1) / d;

        // 求解x的最小正整数解：x的通解为 x0 + k*(a2/d)，k为整数
        // t = a2/d 是x的解的周期，取模后得到最小正整数解
        int t = a2 / d;
        x = (x % t + t) % t; // 保证x是正的最小整数解

        // 更新合并后的方程：
        // 合并后新方程为 x ≡ m1_new (mod a1_new)
        // m1_new = a1*x + m1 （当前合并后的最小解）
        m1 = a1 * x + m1;
        // a1_new = lcm(a1,a2) = |a1*a2| / gcd(a1,a2) （最小公倍数）
        a1 = llabs(a1 / d * a2);
    }

    if (flag)
        printf("-1"); // 无解输出-1
    else
        printf("%lld", (m1 % a1 + a1) % a1); // 输出最小正整数解（保证非负）
    return 0;
}
