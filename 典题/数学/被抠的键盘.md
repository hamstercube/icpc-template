# P14582 [LNCPC 2025] 被抠的键盘


## 题目描述

您有一个键盘，只有 $0,1,2,\cdots,9$ 的数字按键。

对于任意正整数 $m$，显然您总能按出一个为 $m$ 的倍数的**正整数**。

现在，您的 $k$ 个**正整数字**按键被抠掉了，但是您还是想按出一个为 $m$ 的倍数的**正整数**。

## 输入格式

每个测试点包含多组测试数据。第一行给定一个整数 $T(1\leq T\leq  5 \times 10^4)$，表示测试数据组数。

对于每组测试数据：\
第一行给定两个正整数 $m,k(1 \leq m \leq 10^7,0 \leq k \leq 9)$，表示在 $k$ 个按键被抠掉后，您需要按出一个为 $m$ 的倍数的**正整数**。\
第二行给定 $k$ 个互不相同的**正整数** $p_i(1\leq p_i \leq 9)$，表示被抠的按键。

## 输出格式

对于每组测试数据：

如果有解，那么由多次“按 $b$ 次数字 $a$”的操作输出您的操作：\
第一行输出一个整数 $n$，要求 $1\leq n\leq 100$，表示您所需的操作数。\
接下来 $n$ 行，每行输出两个整数 $a,b$，要求 $0 \leq a \leq 9,1 \leq b \leq 10^{18}$，表示一次“按 $b$ 次数字 $a$”的操作。

如果无解，那么输出一行 $-1$。

## 输入输出样例 #1

### 输入 #1

```
2
37 7
2 3 5 6 7 8 9
7 9
1 2 3 4 5 6 7 8 9 
```

### 输出 #1

```
2
1 3
0 1
-1
```

## 说明/提示

对于样例的第一组测试数据，样例的标准输出中的操作按出的正整数是 $1110$，$1110=37\times 30$ 是 $37$ 的一个倍数。

```cpp
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll qpow(ll a, ll b, ll mod){
    ll res = 1;
    a %= mod;
    for(;b;b>>=1){
        if(b&1) res = (__int128)res * a % mod;
        a = (__int128)a * a % mod;
    }
    return res;
}

// 欧拉函数 phi(n)
ll phi(ll n){
    ll res = n;
    for(ll i=2;i*i<=n;i++){
        if(n%i==0){
            while(n%i==0) n/=i;
            res = res / i * (i-1);
        }
    }
    if(n>1) res = res / n * (n-1);
    return res;
}

// 获取x所有正约数，升序
vector<ll> get_div(ll x){
    vector<ll> d;
    for(ll i=1;i*i<=x;i++){
        if(x%i==0){
            d.push_back(i);
            if(i != x/i) d.push_back(x/i);
        }
    }
    sort(d.begin(),d.end());
    return d;
}

// 求10模n的乘法阶，要求gcd(10,n)=1
ll multiplicative_order(ll n){
    ll ph = phi(n);
    vector<ll> divs = get_div(ph);
    for(ll L : divs){
        if(qpow(10, L, n) == 1){
            return L;
        }
    }
    return -1; // 不会到达
}

// d:可用数字，mod = m'，求最小L，使得 d*(10^L‑1)/9 ≡0 (mod mod)
ll find_L(ll d, ll mod){
    if(mod == 1) return 1; // 任何1位d都满足
    ll g = __gcd(d, 9LL * mod);
    ll K = (9LL * mod) / g;
    // gcd(10,K)=1，求10模K阶
    ll L = multiplicative_order(K);
    return L;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int T;
    cin >> T;
    while(T--){
        ll m; int k;
        cin >> m >> k;
        vector<bool> ban(10,false);
        for(int i=0;i<k;i++){
            int p; cin >> p;
            ban[p] = true;
        }
        // 找任意一个可用非0数字d
        int d = -1;
        for(int i=1;i<=9;i++){
            if(!ban[i]){
                d = i; break;
            }
        }
        if(d == -1){ // 1~9全部被抠，无解
            cout << "-1\n";
            continue;
        }
        // 分解 m = 2^c2 * 5^c5 * m_prime
        int c2=0,c5=0;
        ll mp = m;
        while(mp % 2 == 0) { c2++; mp /=2; }
        while(mp %5 ==0) { c5++; mp /=5; }
        int t = max(c2, c5);
        ll L = find_L(d, mp);

        // 输出
        if(t == 0){
            cout << "1\n";
            cout << d << " " << L << "\n";
        }else{
            cout << "2\n";
            cout << d << " " << L << "\n";
            cout << 0 << " " << t << "\n";
        }
    }
    return 0;
}
```

```cpp
#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
typedef unsigned long long ull;

ll n, m;

ll quick(ll x, ll k, ll mod){
    ll res = 1;
    while(k){
        if(k&1) res = res*x%mod;
        k >>= 1;
        x = x*x%mod;
    }
    return res;
}

ll phi(ll x){
    ll res = x;
    for(ll i = 2; i*i <= x; i += 1){
        if(x%i != 0) continue;
        while(x%i == 0) x /= i; 
        res = res/i*(i-1);
    }
    if(x > 1) res = res/x*(x-1); 
    return res;
}

ll g_y(ll x){
    vector<ll> v;
    ll px = phi(x);
    for(ll i = 1; i*i <= px; i++){
        if(px%i != 0) continue;
        v.push_back(i);
        if(i != px/i) v.push_back(px/i);
    }
    sort(v.begin(), v.end());
    for(auto it : v){
        if(quick(10, it, x) == 1){
            return it;
        }
    }
    return 0;
}

void solve(){
    cin >> m >> n;
    vector<ll> can(10, 1);
    for(ll i = 1; i <= n; i++){
        ll x;
        cin >> x;
        can[x] = 0;
    }
    ll d = 1;
    while(d < 10 && can[d] == 0) d++;
    if(d == 10){
        cout << -1 << endl;
        return ;
    }
    while(m%2 == 0) m /= 2;
    while(m%5 == 0) m /= 5;
    ll mt = 9*m/__gcd(d, 9*m);
    cout << 2 << endl;
    cout << d << " " << g_y(mt) << endl;
    cout << 0 << " " << 10000000 << endl;
}

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    ll T = 1;
    cin >> T;
    while(T--) solve();
    return 0;
}
```
