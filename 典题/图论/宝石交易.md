```cpp
#include <bits/stdc++.h>

using namespace std;

vector<vector<int>> g(510, vector<int>(510, 1e9));

void f(){
    for(int k = 1; k <= 410; k ++){
        for(int i = 1; i <= 410; i ++){
            for(int j = 1; j <= 410; j ++){
                g[i][j] = min(g[i][j],g[i][k] + g[k][j]);
            }
        }
    }
}

int main(){
    ios :: sync_with_stdio(0);
    cin.tie(0);
    int n,m;
    cin >> n >> m;
    vector<int> s(n + 10) , t(n + 10);
    for(int i = 1; i <= n; i ++) cin >> s[i];
    for(int i = 1; i <= n; i ++) cin >> t[i];
    while(m --){
        int a,b,c;
        cin >> a >> b >> c;
        g[a][b] = min(g[a][b],c);
    }
    f();
    int ans = 0, minn = 1e10;
    bool flag1 = 0;
    for(int i = 1; i <= n; i ++){
    	ans = 0;
    	bool flag = 0;
        for(int j = 1, h = i; j <= n; j++, h = h!=n? h+1:1){
        	if(s[j] == t[h]) continue;
	        int a1 = g[t[h]][s[j]];
	        int a2 = g[s[j]][t[h]];
	        int qq = min(a1,a2);
	        if(qq >= 1e9){
	        	flag = 1;
	            break;
	        }
	        ans += qq;
	        if(ans >= minn){
	        	flag = 1;
	        	break;
			}
		}
        if(!flag)minn = min(minn, ans), flag1 = 1;
    }
    if(!flag1){
    	cout << -1 << endl;
    	return 0;
	}
    cout << minn << endl;
    return 0;
}
```
