```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
ll dp[110][110], dp1[110][110];
 
int main(){
	memset(dp, 0x3f3f3f, sizeof dp);
	cin >> n >> m; 
	for(ll i = 1; i <= n; i++) dp[i][i] = 0;
	for(ll i = 0; i < m; i++){
		ll t1, t2, t3;
		cin >> t1 >> t2 >> t3;
		dp[t1][t2] = t3;
		dp[t2][t1] = t3;
	}
	for(ll k = 1; k <= n; k++){
		for(ll i = 1; i <= n; i++){
			for(ll j = 1; j <= n; j++){
				dp[i][j] = min(dp[i][j], dp[i][k]+dp[k][j]);
			}
		}
	}
	ll ans = 1e18;
	for(ll u = 1; u <= n; u++){
		for(ll v = u+1; v <= n; v++){
			for(ll i = 1; i <= n; i++){
				for(ll j = 1; j <= n; j++){
					dp1[i][j] = dp[i][j];
				}
			}
			for(ll i = 1; i <= n; i++){
				for(ll j = 1; j <= n; j++){
					dp1[i][j] = min({dp1[i][j], dp1[i][u]+dp1[v][j], dp1[i][v]+dp1[u][j]});
				}
			}
			ll ans1 = 0;
			for(ll i = 1; i <= n; i++){
				for(ll j = i; j <= n; j++){
					ans1 += dp1[i][j];
				}
			}
			ans = min(ans, ans1);
		//	cout << ans << endl;
		}
	}
	cout << ans << endl;
}
```
