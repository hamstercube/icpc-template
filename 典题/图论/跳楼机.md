同余最短路

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m, a[3], p;
priority_queue<ll, vector<ll>, greater<ll>> q; 
map<ll, bool> vis;
vector<ll> d;
ll N = (1ll<<63) -1;

void dij(){
	p = a[2];
	d[0] = 0;
	q.push(0);
	while(q.size()){
		ll x = q.top();
		q.pop();
		if(vis[x%p]) continue;
		vis[x%p] = 1;
		for(ll i = 0; i < 2; i++){
			ll xt = x+a[i];
			if(xt < d[xt%p] && xt <= n){
				d[xt%p] = xt;
				q.push(xt);
			}
		}
	}
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0); 
	cin >> n;
	cin >> a[0] >> a[1] >> a[2];
	d.resize(a[2], N);
	n--;
	dij();
	ll ans = 0;
	for(ll i = 0; i < a[2]; i++){
		if(d[i] != N)
		ans += (n-d[i])/a[2]+1;
		//cout << d[i] << endl;
	}
	cout << ans << endl;
	return 0;
}
```
