线段树加乘操作

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

//建树时记录叶子节点深度 
const ll N = 1e6; 
ll  P = 1, DEP = 0, tr[N+10], add[N+10], mul[N+10];
ll i, j, n, m, v[N+10], mod;

void build(){
  	for(P = 1, DEP = 0; P <= n; P<<=1, DEP++);
  	for(ll i = 1; i <= n; i++) tr[i+P] = v[i];
  	for(ll i = (P+n)>>1; i; i--) tr[i] = tr[i<<1]+tr[i<<1|1], tr[i]%=mod; 
}

void push_down(ll now, ll siz){
	ll lson = now<<1, rson = now<<1|1;
	tr[lson] = tr[lson]*mul[now]%mod+siz/2*add[now]%mod;
	tr[lson] %= mod;
	add[lson] = mul[now]*add[lson]%mod+add[now];
	add[lson] %= mod;
	mul[lson] *= mul[now];
	mul[lson] %= mod; 
	
	tr[rson] = tr[rson]*mul[now]%mod+siz/2*add[now]%mod;
	tr[rson] %= mod;
	add[rson] = mul[now]*add[rson]%mod+add[now];
	add[rson] %= mod;
	mul[rson] *= mul[now]; 
	mul[rson] %= mod;
	
	mul[now] = 1, add[now] = 0;
}

void update_add(ll l,ll r,ll k){
  	l=P+l-1; r=P+r+1;
  	//先下放标记 
  	for(ll i = DEP; i; i--) push_down(l>>i,1<<i),push_down(r>>i,1<<i); 
  	//push_dwon( 链上节点 , 当前子树大小 );
  	int siz = 1;
  	for(; l^1^r; ){
  		if(~l&1) tr[l^1]+=siz*k%mod, add[l^1]+=k, tr[l^1]%=mod, add[l^1]%=mod;//正常更新
  		if(r&1) tr[r^1]+=siz*k%mod, add[r^1]+=k, tr[r^1]%=mod, add[r^1]%=mod;
  		l>>=1; r>>=1; siz<<=1;

  		tr[l] = tr[l<<1]+tr[l<<1|1];//由于标记已下放，所以维护时不再考虑累加标记 
  		tr[r] = tr[r<<1]+tr[r<<1|1];
  		tr[l] %= mod, tr[r] %= mod;
  	}
  	for(l>>=1; l ;l>>=1) tr[l] = tr[l<<1]+tr[l<<1|1], tr[l] %= mod;//上传至根节点 
}

void update_mul(ll l,ll r,ll k){
  	l += P-1, r += P+1;
  	for(ll i = DEP; i; i--) push_down(l>>i,1<<i),push_down(r>>i,1<<i);
  	for(; l^1^r; ){
  		if(~l&1) tr[l^1] = tr[l^1]*k%mod, mul[l^1] = mul[l^1]*k%mod, add[l^1] = add[l^1]*k%mod;//标记覆盖
  		if(r&1) tr[r^1] = tr[r^1]*k%mod, mul[r^1] = mul[r^1]*k%mod, add[r^1] = add[r^1]*k%mod;
  		l>>=1; r>>=1;
  		tr[l] = tr[l<<1]+tr[l<<1|1];
  		tr[r] = tr[r<<1]+tr[r<<1|1];
  		tr[l] %= mod, tr[r] %= mod;
  	}
  	for(l>>=1; l ;l>>=1) tr[l] = tr[l<<1]+tr[l<<1|1], tr[l] %= mod;
}
  
ll query(ll l,ll r){
  	l = l+P-1, r = r+P+1;

  	for(ll i = DEP; i ; i--) push_down(l>>i,1<<i),push_down(r>>i,1<<i); 
  	ll res = 0;
  	for(; l^1^r;){
  		if(~l&1) res += tr[l^1], res %= mod; 
  		if(r&1) res += tr[r^1], res %= mod; 
  		l >>= 1; r >>= 1;
  	}
  	return res;
}

int main(){
	fill(mul, mul+N+1, 1);
	cin >> n >> m >> mod;
	for(ll i = 1; i <= n; i++){
		cin >> v[i];
	}
	build();
	while(m--){
		ll op, l, r, d;
		cin >> op;
		if(op == 1){
			cin >> l >> r >> d;
			update_mul(l, r, d);
		}
		else if(op == 2){
			cin >> l >> r >> d;
			update_add(l, r, d);
		}
		else{
			cin >> l >> r;
			cout << query(l, r) << endl;
		}
	}
	return 0;
}
```
