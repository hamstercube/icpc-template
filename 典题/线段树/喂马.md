处理多次区间变底加权问题

```cpp
#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
typedef unsigned long long ull;

ll i, j, n, m, q, P, DEP;
vector<ll> tr, lz;
ll ans[2000100];
ll v[2000100];

void init(){
    tr.assign(n*3+10, -1);
    lz.assign(n*3+10, 0);
}

void build(){
    init();
  	for(P = 1, DEP = 0; P <= n; P<<=1, DEP++);
  	for(ll i = 1; i <= n; i++) tr[i+P] = v[i];
    for(ll i = n+P>>1; i; i--) {
        tr[i] = tr[i<<1]==tr[i<<1|1] ? tr[i<<1] : -1;
    }
}

void push_down1(ll now, ll siz){ //push_dwon( 链上节点 , 当前子树大小（不是孩子的，孩子的要/2） )
    if(lz[now] == 0) return ;
    ll k = lz[now];
	ll lson = now<<1, rson = now<<1|1;
    ll s_siz = siz/2; // 孩子节点的大小

    if(tr[lson] != -1) ans[tr[lson]] += k*s_siz;
    else lz[lson] += k;
    if(tr[rson] != -1) ans[tr[rson]] += k*s_siz;
    else lz[rson] += k;

	lz[now] = 0;
}

void push_down2(ll now, ll siz){ //push_dwon( 链上节点 , 当前子树大小（不是孩子的，孩子的要/2） )
    if(tr[now] == -1) return ;
    ll k = tr[now];
	ll lson = now<<1, rson = now<<1|1;
    ll s_siz = siz/2; // 孩子节点的大小
    
    tr[lson] = k;
    tr[rson] = k; // lz记得得看情况更改！
}

void push_up(ll i){
    for( ; i; i>>=1) {
        tr[i] = tr[i<<1]==tr[i<<1|1] ? tr[i<<1] : -1;
    }
}

void update1(ll l, ll r, ll k){
  	l=P+l-1; r=P+r+1;
  	//先下放标记 
  	for(ll i = DEP; i; i--) push_down1(l>>i, 1<<i), push_down1(r>>i, 1<<i); 
    for(ll i = DEP; i; i--) push_down2(l>>i, 1<<i), push_down2(r>>i, 1<<i); 
  	ll siz = 1;
  	for(; l^1^r; ){
  		if(~l&1){
            tr[l^1] = k;
        }//正常更新
  		if(r&1){
            tr[r^1] = k;
        }
  		l>>=1; r>>=1; siz<<=1;
        
  		tr[l] = tr[l<<1]==tr[l<<1|1] ? tr[l<<1] : -1;//由于标记已下放，所以维护时不再考虑累加标记 
        tr[r] = tr[r<<1]==tr[r<<1|1] ? tr[r<<1] : -1;
  	}
    push_up(l);
}

void update2(ll l, ll r, ll k){
  	l=P+l-1; r=P+r+1;
  	//先下放标记 
  	for(ll i = DEP; i; i--) push_down1(l>>i, 1<<i), push_down1(r>>i, 1<<i); 
    for(ll i = DEP; i; i--) push_down2(l>>i, 1<<i), push_down2(r>>i, 1<<i); 
  	ll siz = 1;
  	for(; l^1^r; ){
  		if(~l&1){
            if(tr[l^1] != -1) ans[tr[l^1]] += k*siz;
            else lz[l^1] += k;
        }//正常更新
  		if(r&1){
            if(tr[r^1] != -1) ans[tr[r^1]] += k*siz;
            else lz[r^1] += k;
        }
  		l>>=1; r>>=1; siz<<=1;
  	}
}

void get_all(){
    for(ll i = 1, sz = 1<<DEP, j = 1, add = 2; i <= P+n>>1; i++){
        push_down1(i, sz); 
        if(i == j) sz >>= 1, j += add, add <<= 1;
    }
}

void solve(){
    cin >> n >> m >> q;
    for(ll i = 1; i <= n; i++){
        ans[i] = 0;
        cin >> v[i];
    }
    build();
    while(q--){
        ll op, l, r, x;
        cin >> op >> l >> r >> x;
        if(op == 1){
            update1(l, r, x);
        }
        else update2(l, r, x);
    }
    get_all();
    for(ll i = 1; i <= m; i++){
        cout << ans[i] << " ";
    }
}

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    ll T;
    cin >> T;
    while(T--) solve();
    return 0;
}
```
