#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
typedef unsigned long long ull;
mt19937_64 rnd(time(0));

ll i, j, n, m;
deque<array<ll, 2>> de; 

void solve(){
    de.clear();
    cin >> n;
    for(ll i = 1; i <= n; i++){
        ll op;
        cin >> op;
        if(op == 1){
            ll x;
            cin >> x;
            de.push_back({x+1, n});
        } else if(op == 2){
            ll x;
            cin >> x;
            de.push_back({1, x+1});
        } else{
            ll l, r;
            cin >> l >> r;
            de.push_back({l+1, r+1});
        }
    }    
    sort(de.begin(), de.end());
    priority_queue<ll, vector<ll>, greater<ll>> q;
    ll lt = 0;
    for(ll i = 1; i <= n; i++){
        while(de.size() && de.front()[0] <= i){
            q.push({de.front()[1]});
            de.pop_front();
        }
        if(q.size() == 0){
            cout << "N" << endl;
            return ;
        }
        auto r = q.top();
        if(i > r){
            cout << "N" << endl;
            return;
        }  else q.pop();
    }
    cout << "Y" << endl;
}

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    ll T = 1;
    cin >> T;
    while(T--) solve();
    return 0;
}
