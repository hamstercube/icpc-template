志乃的挖坑
前缀和还原等差数列

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;


ll i, j, n, m, k;
vector<array<ll, 2>> v(1); 

bool check(ll mid){
    vector<ll> v1(n+3), v2(n+3);
    for(ll i = 1; i <= mid; i++){
        auto [p, f] = v[i];
        v1[max(1ll, p-f+1)] += 1;
        v1[p+1] += -1;
        v2[min(n, p+f-1)] += 1;
        v2[p] += -1;
    }
    for(ll i = 1; i <= n; i++){
        v1[i] += v1[i-1];
        v2[n-i+1] += v2[n-i+2];
    }
    for(ll i = 1; i <= mid; i++){
        auto [p, f] = v[i];
        if(p-f+1 < 1) v1[1] += -(p-f+1)+1;
        if(p+f-1 > n) v2[n] += p+f-1-n;
        v1[p+1] += -f;
        v2[p] += -f+1;
    }
    for(ll i = 1; i <= n; i++){
        v1[i] += v1[i-1];
        v2[n-i+1] += v2[n-i+2];
    }
    for(ll i = 1; i <= n; i++){
        if(v1[i]+v2[i] > k) return 1;
    }
    return 0;
}

int main(){
    cin >> n >> m >> k;
    for(ll i = 1; i <= m; i++){
        ll x, y;
        cin >> x >> y;
        v.push_back({x, y});
    }
    ll l = 1, r = m;
    ll ans = -1;
    while(l <= r){
        ll mid = l+r>>1;
        if(check(mid)){
            r = mid-1;
            ans = mid;
        }else l = mid+1;
    }
    if(ans == -1) cout << "No";
    else{
        cout << "Yes" << endl << ans;
    }
    return 0;
}
```
