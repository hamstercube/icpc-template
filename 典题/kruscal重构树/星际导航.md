处理最优路径最小值或最大值

```cpp
#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;

ll i, j, n, m, sz, k;
ll fc[200100][25], f[200100], dep[200100], val[200100];
vector<ll> road[200100];
vector<pair<ll, pair<ll, ll>>> v;

ll Find(ll x){
	if(f[x] == x) return x;
	return f[x] = Find(f[x]);
}

void kruskal_G(){
	sort(v.begin(), v.end());
	ll cot = 0;
	for(ll i = 0; i < m; i++){
		auto v1 = v[i].first;
		auto [x, xt] = v[i].second;
		ll f1 = Find(x), f2 = Find(xt);
		if(f1 == f2) continue;
		f[f1] = f[f2] = ++sz;
		road[sz].push_back(f1), road[sz].push_back(f2);
		val[sz] = v1;
		cot++;
	}
}

void dfs(ll x, ll fa){
	dep[x] = dep[fa]+1, fc[x][0] = fa;
	for(ll i = 1; i <= 20; i++){
		fc[x][i] = fc[fc[x][i-1]][i-1];
	}
	for(auto xt : road[x]){
		if(xt == fa) continue;
		dfs(xt, x);
	}
}

ll lca(ll a, ll b){
	if(dep[a] < dep[b]) swap(a, b);
	for(ll i = 20; i >= 0; i--){
		if(dep[fc[a][i]] >= dep[b]) a = fc[a][i];
	}
	if(a == b) return a;
	for(ll i = 20; i >= 0; i--){
		if(fc[a][i] != fc[b][i]){
			a = fc[a][i], b = fc[b][i];
		}
	}
	return fc[a][0];
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0); 
	for(ll i = 1; i <= 2e5; i++) f[i] = i; // 初始化所有集合 !!!!! 要记得初始化n以外的点 
	cin >> n >> m;
	sz = n;
	for(ll i = 0; i < m; i++){
		ll t1, t2, t3;
		cin >> t1 >> t2 >> t3;
		v.push_back({t3, {t1, t2}});
	}
	kruskal_G();
	for(ll i = sz; i > n; i--) if(!dep[i]) dfs(i, 0);
	cin >> k;
	while(k--){
		ll t1, t2;
		cin >> t1 >> t2;
		if(Find(t1) != Find(t2)) cout << "impossible" << endl;
		else cout << val[lca(t1, t2)] << endl;
	}
	return 0;
}
```
