#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<pair<ll, ll>>> q; 

void solve(){
	while(q.size()) q.pop();
	cin >> n;
	vector<ll> dp(n+5, 0), v(n+5, 0), lm(n+5, 0);
	for(ll i = 1; i <= n; i++) cin >> v[i];
	cin >> m;
	for(ll i = 0; i < m; i++){
		ll l, r;
		cin >> l >> r;
		lm[r+1] = max(lm[r+1], l);
	}
	for(ll i = 2; i <= n+1; i++){
		lm[i] = max(lm[i], lm[i-1]);
	}
	q.push({0, 0});
	ll ans = 0;
	for(ll i = 1; i <= n+1; i++){
		while(q.size() && q.top().second < lm[i]) q.pop();
		dp[i] = v[i]+q.top().first;
		q.push({dp[i], i});	
	}
	cout << dp[n+1] << endl;
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	ll T;
	cin >> T;
	while(T--) solve();
	return 0;
}
