n(logn) 求上升子序列

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

ll i, j, n, m;
ll pre[200100], suf[200100];
ll v[200100];

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    cin >> n;
    for(ll i = 1; i <= n; i++){
        cin >> v[i];
    }
    auto get_l = [&]() -> ll{
        deque<ll> de;
        for(ll i = 1; i <= n; i++){
            if(de.size() == 0 || de.back() < v[i]){
                de.push_back(v[i]);
                pre[i] = de.size();
            } else{
                ll p = lower_bound(de.begin(), de.end(), v[i])-de.begin();
                de[p] = v[i];
                pre[i] = p+1;
            }
        }
        return (ll) de.size();
    };
    ll len = get_l();
    deque<ll> de[2];
    cout << len << endl;
    for(ll i = 1, j = 1; i <= n; i++){
        if(j == pre[i]) de[0].push_back(v[i]), j++;
        if(de[0].size() == len) break;
    }
    for(ll i = n, j = len; i >= 1; i--){
        if(pre[i] == j) de[1].push_front(v[i]), j--;
        if(de[1].size() == len) break;
    }
    for(ll i = 0; i < 2; i++){
        for(auto it : de[i]) cout << it << " ";
        cout << endl;
    }
    return 0;
}
```
