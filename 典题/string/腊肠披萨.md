字符串哈希，计数去重子串

```cpp
#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
typedef unsigned long long ull;

ll i, j, n, m, c;
ull P = 13331ull;
const ll N = 3000000;
unordered_map<ull, ll> cot[N+1];
ll mod;
string v[N+10];
ll val[N+10], ne[N+10];
string s;

void init(){
    val[0] = 1;
    for(int i = 1; i <= N; i++){
        val[i] = val[i-1]*c%mod;
    }
}

void solve(){
    cin >> n >> c >> mod;
    init();
    for(int i = 1; i <= n; i++){
        cin >> v[i];
        v[i] = " "+v[i];
        cot[0][0]++;
        ull key = 0, k = 1;
        for(int j = v[i].size()-1, len = 1; j >= 1; j--, k *= P, len++){
            key += k*v[i][j];
            cot[len][key]++;
        }
    }
    ll ans = 0;
    for(int i = 1; i <= n; i++){
        ull key = 0;
        ne[1] = 1;
        for(int j = 2, p = 1; j < v[i].size(); j++){
            while(v[i][j] != v[i][p] && p != 1) p = ne[p-1];
            if(v[i][j] == v[i][p]) p++;
            ne[j] = p;
        }
        ans = (ans+n)%mod;
        for(int j = 1; j < v[i].size(); j++){
            key = key*P+v[i][j];
            if(cot[j].count(key) == 0) continue; 
            ans = (ans+cot[j][key]*val[j]%mod)%mod;
            ans = (ans-val[ne[j]-1]*cot[j][key]%mod+mod)%mod;
        }
    }
    cout << ans << endl;
}


int main(){
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    ll T = 1;
    while(T--) solve();
    return 0;
}
```
