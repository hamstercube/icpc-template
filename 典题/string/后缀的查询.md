字典树换根

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 2e6 + 10;
ll i, j, n, m;
ll ch[N][26];
ll cot[N];
ll sz = 1;

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    cin >> n;
    ll op;
    string s;
    ll rt = 1;
    auto update = [&](string &s){
        ll p = rt;
        cot[p]++;
        for(ll i = s.size()-1; i >= 0; i--){
            ll t = s[i]-'a';
            if(ch[p][t] == 0) ch[p][t] = ++sz;
            p = ch[p][t];
            cot[p]++;
        }
    };
    auto change = [&](char &c){
        ll t = c-'a';
        ll p = ++sz;
        ch[p][t] = rt;
        cot[p] = cot[rt];
        rt = p;
    };
    auto query = [&](string &s) -> ll{
        ll p = rt, res = 0;
        for(ll i = s.size()-1; i >= 0; i--){
            ll t = s[i]-'a';
            if(ch[p][t] == 0) break;
            p = ch[p][t];
            res += cot[p];
        }
        return res;
    };
    while(n--){
        cin >> op;
        if(op == 1){
            char c;
            cin >> c;
            change(c);
        }else if(op == 2){
            cin >> s;
            update(s);
        }else{
            cin >> s;
            cout << query(s) << '\n';
        }
    }
    return 0;
}
```
