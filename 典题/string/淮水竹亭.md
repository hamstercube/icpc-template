```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m, sz = 0;
ll tree[1000100][26], cot[1000100];
ll ne[1000100];
ll dp[3010][3010];
ll mod = 998244353;

void update(string &s){
	ll p = 0;
	for(ll i = 0; i < s.size(); i++){
		ll t = s[i]-'a';
		if(!tree[p][t]) tree[p][t] = ++sz;
		p = tree[p][t];
	}
	cot[p]++;
}

void build(){
	queue<ll> q;
	for(ll i = 0; i < 26; i++){
		if(tree[0][i]) q.push(tree[0][i]);
	}
	while(q.size()){
		ll u = q.front();
		q.pop();
        cot[u] |= cot[ne[u]];
		for(ll i = 0; i < 26; i++){
			ll v = tree[u][i];
			if(v) ne[v] = tree[ne[u]][i], q.push(v);
			else tree[u][i] = tree[ne[u]][i];
		}
	}
}

void solve(){
	cin >> n >> m;
	for(i = 0; i < n; i++){
		string s;
		cin >> s;
		update(s);
	}
	build();
    dp[0][0] = 1;
	for(ll len = 0; len < m; len++){
        for(ll i = 0; i <= sz; i++){
            if(dp[len][i] == 0) continue;
            for(ll j = 0; j < 26; j++){
                ll p = tree[i][j];
                if(cot[p] == 0) dp[len+1][p] = (dp[len+1][p]+dp[len][i])%mod; 
            }
        }
    }
    ll ans = 0;
    for(ll i = 0; i <= sz; i++){
        ans = (ans+dp[m][i])%mod;
    }
    cout << ans;
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	ll T = 1;
	while(T--){
		solve();
	}
	return 0;
}
```
