#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
double a[1010], b[1010], v[1010];
double lim = 1e-6;

bool cmp(double a, double b){
	return a > b;
}

bool check(double x){ //根据题意修改
	double sum = 0;
	for(ll i = 0; i < n; i++){
		v[i] = a[i]-x*b[i];
	}
	sort(v, v+n, cmp);
	for(ll i = 0; i < m; i++) sum += v[i];
	if(sum >= 0) return 1;
	return 0;
}

int main(){
	while(1){
		cin >> n >> m;
		m = n-m;
		if(n == m && !n) return 0;
		double l = 0, r = 0;
		for(ll i = 0; i < n; i++) cin >> a[i], r += a[i];
		for(ll i = 0; i < n; i++) cin >> b[i];
		double res = 0;
		while(r-l >= lim){
			double mid = (l+r+lim)/2;
			if(check(mid)){
				l = mid;
			}
			else r = mid-lim;
		}
		ll ans = l*100+0.5;
		cout << ans << endl;
	}
	return 0;
}
