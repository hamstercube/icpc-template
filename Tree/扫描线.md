#include<bits/stdc++.h>

typedef long long ll;

ll i, j, n, m, nt;
vector<array<ll, 4>> line;
vector<ll> v(1), vv(1);
ll tr[1001000], cnt[1000100], val[1000100];
ll P;

void build(){
    for(P = 1; P <= nt; P<<=1);
    for(ll i = 1; i <= nt; i++) val[i+P] = vv[i];
    for(ll i = (nt+P)>>1; i; i--) val[i] = val[i<<1] + val[i<<1|1];
}

void push_up(ll l){
    for( ; l; l>>=1){
        if(cnt[l]) tr[l] = val[l];
        else tr[l] = tr[l<<1] + tr[l<<1|1];
    }
}

void update(ll l, ll r, ll d){
    l = l+P-1, r = r+P+1;
    for( ; l^1^r; ){
        if(~l&1){
            cnt[l^1] += d;
            if(cnt[l^1]) tr[l^1] = val[l^1];
            else tr[l^1] = tr[(l^1)<<1]+tr[(l^1)<<1|1];
        } 
        if(r&1){
            cnt[r^1] += d;
            if(cnt[r^1]) tr[r^1] = val[r^1];
            else tr[r^1] = tr[(r^1)<<1]+tr[(r^1)<<1|1];
        }
        l >>= 1, r >>= 1;
        if(cnt[l]) tr[l] = val[l];
        else tr[l] = tr[l<<1] + tr[l<<1|1];
        if(cnt[r]) tr[r] = val[r];
        else tr[r] = tr[r<<1] + tr[r<<1|1];
    }
    push_up(l>>1);
}

ll query(ll l, ll r){
    if(r < l) return 0;
    l = l+P-1, r = r+P+1;
    ll res = 0;
    for( ; l^1^r; ){
        if(~l&1){
            res += tr[l^1];
        } 
        if(r&1){
            res += tr[r^1];
        }
        l >>= 1, r >>= 1;
    }
    return res;
}

        line.push_back({x1, y1, y2, 1});
        line.push_back({x2, y1, y2, -1});
    }
    sort(v.begin()+1, v.end());
    v.erase(unique(v.begin()+1, v.end()), v.end());
    sort(line.begin(), line.end());
    for(ll i = 1; i < v.size()-1; i++){
        vv.push_back(v[i+1]-v[i]);
    }
    nt = vv.size()-1;
    build();
    ll ans = 0;
    for(ll i = 0; i < line.size(); i++){
        auto [x, l, r, d] = line[i];
        
        if(i > 0){
            auto [px, pl, pr, pd] = line[i-1];

        }

        ll rt = lower_bound(v.begin()+1, v.end(), r)-v.begin();
        // cout << l << " " << r << " " << d << endl;

        update(lt, rt-1, d);
    }

}

  		if(~l&1){

            else if(siz == 1) tr[l^1] = 0;

        }
  		if(r&1){
            cnt[r^1] += k;
  			if(cnt[r^1]) tr[r^1] = siz;
            else if(siz == 1) tr[r^1] = 0;
            else tr[r^1] = tr[(r^1)<<1]+tr[(r^1)<<1|1]; 
        }
        
  		l >>= 1, r >>= 1, siz <<= 1;
  		
        if(cnt[l]) tr[l] = siz;
        else if(siz == 1) tr[l] = 0;
    	else tr[l] = tr[l<<1]+tr[l<<1|1]; 
        if(cnt[r]) tr[r] = siz;
        else if(siz == 1) tr[r] = 0;
        else tr[r] = tr[r<<1]+tr[r<<1|1]; 
  	}
    push_up(l>>1, siz<<1);
}
  
ll query(ll l, ll r){
  	l = l+P-1, r = r+P+1;
  	ll res = 0;
  	for( ; l^1^r; ){
  		if(~l&1){
            res += tr[l^1];
        }
  		if(r&1){
            res += tr[r^1];
        }
  		l >>= 1; r >>= 1;
  	}
  	return res;
}

vector<array<ll, 4>> link;

void solve(){
	cin >> n >> k;
	for(ll i = 1; i <= 1e6; i++) de[i].push_back(0);
	for(ll i = 1; i <= n; i++){
		cin >> v[i];
		de[v[i]].push_back(i);

	build();
	for(ll i = 1; i <= 1e6; i++){
		if(de[i].size()-2 < k) continue;
		for(ll j = k; j < de[i].size()-1; j++){
			link.push_back({de[i][j], de[i][j-k]+1, de[i][j-k+1], 1});
			link.push_back({de[i][j+1], de[i][j-k]+1, de[i][j-k+1], -1});
		}
	}
	sort(link.begin(), link.end());
	ll ans = n*(n+1)/2;
	for(ll i = 1, lt = 0; i <= n; i++){
		while(lt < link.size() && link[lt][0] == i){
			auto [ti, l, r, d] = link[lt];
			update(l, r, d);
			lt++; 
		}
		ans -= query(1, n);
	}
	cout << ans << endl;
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0);
	ll T = 1;
	while(T--) solve();
