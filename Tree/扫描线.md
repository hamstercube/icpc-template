#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;

ll i, j, n, m, nt;
vector<array<ll, 4>> line;
vector<ll> v(1), vv(1);
ll tr[1001000], cnt[1000100], val[1000100];
ll P;

void build(){
    for(P = 1; P <= nt; P<<=1);
    for(ll i = 1; i <= nt; i++) val[i+P] = vv[i];
    for(ll i = (nt+P)>>1; i; i--) val[i] = val[i<<1] + val[i<<1|1];
}

void push_up(ll l){
    for( ; l; l>>=1){
        if(cnt[l]) tr[l] = val[l];
        else tr[l] = tr[l<<1] + tr[l<<1|1];
    }
}

void update(ll l, ll r, ll d){
    l = l+P-1, r = r+P+1;
    for( ; l^1^r; ){
        if(~l&1){
            cnt[l^1] += d;
            if(cnt[l^1]) tr[l^1] = val[l^1];
            else tr[l^1] = tr[(l^1)<<1]+tr[(l^1)<<1|1];
        } 
        if(r&1){
            cnt[r^1] += d;
            if(cnt[r^1]) tr[r^1] = val[r^1];
            else tr[r^1] = tr[(r^1)<<1]+tr[(r^1)<<1|1];
        }
        l >>= 1, r >>= 1;
        if(cnt[l]) tr[l] = val[l];
        else tr[l] = tr[l<<1] + tr[l<<1|1];
        if(cnt[r]) tr[r] = val[r];
        else tr[r] = tr[r<<1] + tr[r<<1|1];
    }
    push_up(l>>1);
}

ll query(ll l, ll r){
    if(r < l) return 0;
    l = l+P-1, r = r+P+1;
    ll res = 0;
    for( ; l^1^r; ){
        if(~l&1){
            res += tr[l^1];
        } 
        if(r&1){
            res += tr[r^1];
        }
        l >>= 1, r >>= 1;
    }
    return res;
}

void solve(){
    cin >> n;
    for(ll i = 1; i <= n; i++){
        ll x1, y1, x2, y2;
        cin >> x1 >> y1 >> x2 >> y2;
        v.push_back(y1);
        v.push_back(y2);
        line.push_back({x1, y1, y2, 1});
        line.push_back({x2, y1, y2, -1});
    }
    sort(v.begin()+1, v.end());
    v.erase(unique(v.begin()+1, v.end()), v.end());
    sort(line.begin(), line.end());
    for(ll i = 1; i < v.size()-1; i++){
        vv.push_back(v[i+1]-v[i]);
    }
    nt = vv.size()-1;
    build();
    ll ans = 0;
    for(ll i = 0; i < line.size(); i++){
        auto [x, l, r, d] = line[i];
        
        if(i > 0){
            auto [px, pl, pr, pd] = line[i-1];
            ll ys = query(1, nt);
            // cout << ys << endl;
            ans += (x-px)*ys;
        }
        ll lt = lower_bound(v.begin()+1, v.end(), l)-v.begin();
        ll rt = lower_bound(v.begin()+1, v.end(), r)-v.begin();
        // cout << l << " " << r << " " << d << endl;
        // cout << lt << " " << rt-1 << endl;
        update(lt, rt-1, d);
    }
    cout << ans << endl;
}

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    ll T = 1;
    while(T--) solve();
    return 0;
}
