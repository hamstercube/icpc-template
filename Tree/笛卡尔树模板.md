// 处理RMQ问题， 处理上升子序列问题
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

deque<pair<ll, ll>> st;
vector<array<ll, 2>> tr;
ll n;

int main()
{
	ios::sync_with_stdio(0);
	cin.tie(0);
	cin >> n;
	tr.resize(n+1);
	for (ll i = 1; i <= n; i++){  //建树
		ll a;
		cin >> a;
		ll x=0;
		while (st.size() && a < st.back().second){
			x = st.back().first;
			st.pop_back();
		}
		if (!st.empty()){
			tr[st.back().first][1]=i;
		}
		tr[i][0]=x;
		st.push_back({i,a});
	}
	return 0;
}
