#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m, a[200100];
vector<ll> prime;
bool vis[110];
unordered_map<ll, ll> mp;
ll X[200100][30], pos[200100][30];

void ou_shi(){
    for(ll i = 2; i <= 100; i++){
        if(!vis[i]){
            vis[i] = 1;
            prime.push_back(i);
        }
        for(ll j = 0; j < prime.size() && prime[j]*i <= 100; j++){
            vis[prime[j]*i] = 1;
            if(i%prime[j] == 0) break;
        }
    }
}

void yu(){
    ou_shi();
    for(ll i = 0, j = 1; i < prime.size(); i++, j *= 2) mp[prime[i]] = j;
}

void add(ll x, ll now){
    for(ll i = 0; i < 26; i++){
        X[now][i] = X[now-1][i];
        pos[now][i] = pos[now-1][i];
    }
    ll p = now;
    for(ll i = 25; i >= 0; i--){
        ll t = x>>i;
        if(!t) continue;
        if(!X[now][i]){
            X[now][i] = x;
            pos[now][i] = p;
            return;
        }
        if(p > pos[now][i]){
            swap(p, pos[now][i]);
            swap(x, X[now][i]);
        }
        x ^= X[now][i];
    }
}

bool check(ll l, ll r){
    ll cot = 0;
    for(ll i = 0; i < 26; i++){
        if(pos[r][i] >= l) cot++;
    }
    if(cot == r-l+1) return false;
    return true;
}

int main(){
    ios::sync_with_stdio(0);
    cin.tie(0), cout.tie(0);
    yu();
    cin >> n >> m;
    for(ll i = 1; i <= n; i++){
        ll t;
        cin >> t;
        for(auto it : prime){
            while(t%it == 0){
                a[i] ^= mp[it];
                t /= it;
            }
        }
        add(a[i], i);
    }
    while(m--){
        ll l, r;
        cin >> l >> r;
        if(check(l, r)) cout << "Yes" << endl;
        else cout << "No" << endl;
    }
    return 0;
}
