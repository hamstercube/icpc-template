约瑟夫环

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
ll ans[110];
ll vis[110][110];

ll Find(ll x, ll p){
	vis[p][x] = 1;
	for(ll i = p+1; i <= n; i++){
		x = (x+m-1)%i+1;
		vis[i][x] = 1;
	}
	return x;
}

int main(){
	cin >> n >> m;
	for(ll i = 1; i <= n; i++){
		for(ll j = 1; j <= i; j++){
			if(!vis[i][j]){
				ans[i] = Find(j, i);
				break;
			}
		}
	}
	for(ll i = n; i >= 1; i--){
		cout << ans[i] << " ";
	}
	return 0;
}
```
