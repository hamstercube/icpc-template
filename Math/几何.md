```cpp
struct Point {
    double x, y;
    constexpr Point(double _x = 0, double _y = 0) : x(_x), y(_y) {}
    constexpr Point operator+() const noexcept { return *this; }
    constexpr Point operator-() const noexcept { return Point(-x, -y); }
    constexpr Point operator+(const Point &p) const { return Point(x + p.x, y + p.y); }
    constexpr Point operator-(const Point &p) const { return Point(x - p.x, y - p.y); }
    constexpr Point &operator+=(const Point &p) { return x += p.x, y += p.y, *this; }
    constexpr Point &operator-=(const Point &p) { return x -= p.x, y -= p.y, *this; }
    constexpr double operator*(const Point &p) const { return x * p.x + y * p.y; }
    constexpr Point &operator*=(const double &k) { return x *= k, y *= k, *this; }
    constexpr Point operator*(const double &k) { return Point(x * k, y * k); }
    constexpr bool operator==(const Point &r) const noexcept { return r.x == x && r.y == y; }
    constexpr double cross(const Point &r) const { return x * r.y - y * r.x; }
    bool operator<(const Point &r) const { return std::make_pair(x, y) < std::make_pair(r.x, r.y); }

//Line的定义

struct Line {
    Point a, b;
    Line() = default;
    Line(Point a, Point b) : a(a), b(b) {}
    // ax + by = c
    Line(double A, double B, double C) {
        if(eq(A,0)) {
            a = Point(0, C / B), b = Point(1, C / B);
        } else if(eq(B, 0)) {
            a = Point(C / A, 0), b = Point(C / A, 1);
        } else {
            a = Point(0, C / B), b = Point(C / A, 0);
        }
    }

    // 1 : left, 0 : same, -1 : right 判断一个点 r 在直线 ab 的 左侧、线上、还是在右侧
    //可以算两个点,相乘异号不在用一侧,同号在一侧
    constexpr int toleft(const Point &r) const {
        auto t = (b - a).cross(r - a);
        return t > eps ? 1 : t < -eps ? -1 : 0;
    }


    //输出流(debug)
    friend std::ostream &operator<<(std::ostream &os, Line &ls) {
        return os << "{"
                  << "(" << ls.a.x << ", " << ls.a.y << "), (" << ls.b.x << ", " << ls.b.y << ")}";
    }
};

//三点关系
int ccw(const Point &a, Point b, Point c) {
    b = b - a, c = c - a;

    if(cross(b, c) > 0) return +1;   // 逆时针方向
    if(cross(b, c) < 0) return -1;   // 顺时针方向

    if(dot(b, c) < 0) return +2;     // C 在 A 的后面（共线）
    if(abs2(b) < abs2(c)) return -2; // C 在 B 的前面（共线）
    
    return 0;                        // C 在线段 AB 上
}

//两线段相交
bool intersect(const Segment &s, const Segment &t) { return ccw(s.a, s.b, t.a) * ccw(s.a, s.b, t.b) <= 0 && ccw(t.a, t.b, s.a) * ccw(t.a, t.b, s.b) <= 0; }
// 两多边形相交
bool intersect(const Polygon &ps, const Polygon &qs) {
    int pl = si(ps), ql = si(qs), i = 0, j = 0;
    while((i < pl or j < ql) and (i < 2 * pl) and (j < 2 * ql)) {
        auto ps0 = ps[(i + pl - 1) % pl], ps1 = ps[i % pl];
        auto qs0 = qs[(j + ql - 1) % ql], qs1 = qs[j % ql];
        if(intersect(Segment(ps0, ps1), Segment(qs0, qs1))) return true;
        Point a = ps1 - ps0;
        Point b = qs1 - qs0;
        double v = cross(a, b);
        double va = cross(qs1 - qs0, ps1 - qs0);
        double vb = cross(ps1 - ps0, qs1 - ps0);
 
        if(!v and va < 0 and vb < 0) return false;
        if(!v and !va and !vb) {
            i += 1;
        } else if(v >= 0) {
            if(vb > 0)
                i += 1;
            else
                j += 1;
        } else {
            if(va > 0)
                j += 1;
            else
                i += 1;
        }
    }
    return false;
}
```

```cpp
bool is_convex(const Points &p) {
    int n = (int)p.size();
    for(int i = 0; i < n; i++) {
        if(ccw(p[(i + n - 1) % n], p[i], p[(i + 1) % n]) == -1) return false;
    }
    return true;
}

Points convex_hull(Points p) {
    int n = p.size(), k = 0;
    if(n <= 2) return p;
    sort(begin(p), end(p), [](pt x, pt y) { return (x.x != y.x ? x.x < y.x : x.y < y.y); });
    Points ch(2 * n);
    for(int i = 0; i < n; ch[k++] = p[i++]) {
        while(k >= 2 && cross(ch[k - 1] - ch[k - 2], p[i] - ch[k - 1]) <= 0) --k;
    }
    for(int i = n - 2, t = k + 1; i >= 0; ch[k++] = p[i--]) {
        while(k >= t && cross(ch[k - 1] - ch[k - 2], p[i] - ch[k - 1]) <= 0) --k;
    }
    ch.resize(k - 1);
    return ch;
}
```

```cpp
enum { _OUT, _ON, _IN }; // 0 1 2
//O(n)
int contains(const Polygon &Q, const Point &p) {
    bool in = false;
    for(int i = 0; i < Q.size(); i++) {
        Point a = Q[i] - p, b = Q[(i + 1) % Q.size()] - p;
        if(a.y > b.y) swap(a, b);
        if(a.y <= 0 && 0 < b.y && cross(a, b) < 0) in = !in;
        if(eq(cross(a, b), 0) && dot(a, b) <= eps) return _ON;
    }
    return in ? _IN : _OUT;
}


// -1 : on, 0 : out, 1 : in
// O(log(n))
int is_in(const Polygon &p, const Point &a) {
    if(p.size() == 1) return a == p[0] ? -1 : 0;
    if(p.size() == 2) return intersect(Segment(p[0], p[1]), a);
    if(eq(a.x,p[0].x) && eq(a.y,p[0].y)) return -1;
    if((p[1] - p[0]).toleft(a - p[0]) == -1 || (p.back() - p[0]).toleft(a - p[0]) == 1) return 0;
    const auto cmp = [&](const Point &u, const Point &v) { return (u - p[0]).toleft(v - p[0]) == 1; };
    const size_t i = lower_bound(p.begin() + 1, p.end(), a, cmp) - p.begin();
    if(i == 1) return intersect(Segment(p[0], p[i]), a) ? -1 : 0;
    if(i == p.size() - 1 && intersect(Segment(p[0], p[i]), a)) return -1;
    if(intersect(Segment(p[i - 1], p[i]), a)) return -1;
    return (p[i] - p[i - 1]).toleft(a - p[i - 1]) > 0;
}
```

```cpp
Point circumcenter(const Point &a, const Point &b, const Point &c) {
    Line u, v;
    // 边 AB 的中垂线
    u.a = Point((a.x + b.x) / 2.0, (a.y + b.y) / 2.0);
    u.b = Point(u.a.x - a.y + b.y, u.a.y + a.x - b.x);
    // 边 AC 的中垂线
    v.a = Point((a.x + c.x) / 2.0, (a.y + c.y) / 2.0);
    v.b = Point(v.a.x - a.y + c.y, v.a.y + a.x - c.x);
    // 两条中垂线的交点就是外心
    return crosspoint(u, v);
}
```

```cpp
Point incenter(const Point &a, const Point &b, const Point &c) {
    Line u, v;
    double m, n;

    // 构造顶点A的角平分线
    u.a = a;
    m = atan2(b.y - a.y, b.x - a.x);
    n = atan2(c.y - a.y, c.x - a.x);
    u.b.x = u.a.x + cos((m + n) / 2.0);
    u.b.y = u.a.y + sin((m + n) / 2.0);

    // 构造顶点B的角平分线
    v.a = b;
    m = atan2(a.y - b.y, a.x - b.x);
    n = atan2(c.y - b.y, c.x - b.x);
    v.b.x = v.a.x + cos((m + n) / 2.0);
    v.b.y = v.a.y + sin((m + n) / 2.0);

    // 两条角平分线的交点就是内心
    return crosspoint(u, v);
}
```

```cpp
// 判断线段与圆是否相交（相切 / 端点在圆上 也算相交）
bool segment_circle_intersect(const Segment &s, const Circle &c) {
    // 1. 端点在圆内或圆上
    double da = dot(s.a - c.p, s.a - c.p);
    double db = dot(s.b - c.p, s.b - c.p);
    double r2 = c.r * c.r;

    if (da <= r2 + eps) return true;
    if (db <= r2 + eps) return true;

    // 2. 求圆心在线段上的投影
    Point pr = projection(s, c.p);

    // 3. 投影不在线段上 → 不相交
    if (!intersect(s, pr)) return false;

    // 4. 投影到圆心的距离 ≤ 半径
    double d2 = dot(pr - c.p, pr - c.p);
    return d2 <= r2 + eps;
}
```

```cpp
int circle_circle_relation(const Circle &c1, const Circle &c2) {
    double dx = c1.p.x - c2.p.x;
    double dy = c1.p.y - c2.p.y;
    double d2 = dx * dx + dy * dy; // 圆心距平方
    double d = sqrt(d2);           // 圆心距
    double r1 = c1.r, r2 = c2.r;
    
    if (d > r1 + r2 + eps) return -1;    // 外离
    if (eq(d, r1 + r2)) return 0;        // 外切
    if (d + min(r1, r2) < max(r1, r2) - eps) return 3; // 内含
    if (eq(d + min(r1, r2), max(r1, r2))) return 2;    // 内切
    return 1; // 相交
}

// 简单判断：是否相交（含相切、重合）
bool circle_circle_intersect(const Circle &c1, const Circle &c2) {
    int rel = circle_circle_relation(c1, c2);
    return rel >= 0 && rel <= 2;
}
```

 求圆 c 上到点 p 最近的点
```cpp
Point circle_closest_point(const Circle &c, const Point &p) {
    // 向量：圆心 -> p
    Point vec = p - c.p;
    double d = sqrt(dot(vec, vec)); // 圆心到 p 的距离

    // 如果 p 和圆心重合，返回任意点（这里返回圆心右侧 r 处）
    if (eq(d, 0)) {
        return Point(c.p.x + c.r, c.p.y);
    }

    // 单位化向量 * 半径 + 圆心 = 圆上最近点
    Point unit = vec / d;
    return c.p + unit * c.r;
}
```
