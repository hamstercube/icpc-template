求长度为n的合法括号个数

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m, k;
ll p = 1e9+7;

ll quick(ll x, ll k){
	ll res = 1;
	while(k){
		if(k%2) res = x*res%p;
		x = x*x%p;
		k /= 2;
	}
	return res%p;
}

ll C(ll n, ll m){
	if(m > n-m) m = n-m;
	ll a = 1, b = 1;
	for(i = 0; i < m; i++){
		a = a*(n-i)%p;
		b = b*(i+1)%p;
	}
	return a*quick(b, p-2)%p;
}

int main(){
	cin >> n >> k;
	ll ans = (C(2*n, n)*quick(n+1, p-2)%p)*quick(k, n)%p;
	cout << ans << endl;
}
```
