```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long LL;

constexpr int N = 2e5 + 10, MOD = 1e9 + 7;

const double eps = 1e-9;
int dcmp(double x)
{
    return (fabs(x) <= eps) ? 0 : (x < 0 ? -1 : 1);
}

struct Point 
{
    LL x, y, ang;

    friend Point operator-(const Point &a, const Point &b) {
        return {a.x - b.x, a.y - b.y};
    }
    // 点乘
    friend LL operator*(const Point &a, const Point &b) {
        return a.x * b.x + a.y * b.y;
    }

    // 叉乘
    friend LL operator^(const Point &a, const Point &b) {
        return a.x * b.y - a.y * b.x;
    }
} p[N];

LL sqr(int a, int b, int c) { return abs((p[b] - p[a]) ^ (p[c] - p[b])); } //三个点围成的三角形面积

LL dis(Point p1, Point p2) //两点距离
{
    return sqrt((p1.x - p2.x) * (p1.x - p2.x) + (p1.y - p2.y) * (p1.y - p2.y));
}

LL dis2(Point p1, Point p2) //两点距离
{
    return (p1.x - p2.x) * (p1.x - p2.x) + (p1.y - p2.y) * (p1.y - p2.y);
}

bool cmp(Point p1, Point p2) 
{
    if (p1.ang == p2.ang) return dis(p1, p[1]) < dis(p2, p[1]);
    return p1.ang < p2.ang;
}

LL Cross(Point p1, Point p2) 
{ 
    return p1 ^ p2; 
}

LL dis_PL(Point p, Point A, Point B)//点到直线 AB 的距离 
{
    return abs(Cross(p - A, p - B) / dis(A, B));//面积除以底 
}

void solve()
{
    LL n, xc, yc, rc;
    cin >> n >> xc >> yc >> rc;

    Point o = {xc, yc};

    for (int i = 1; i <= n; i ++) cin >> p[i].x >> p[i].y;

    auto check = [&](int i, int j) -> bool
    {
        LL d = dis2(p[i], p[j]);
        LL cross = Cross((p[i] - p[j]), (p[i] - o));
        if (cross <= 0 || (__int128_t)cross * cross < (__int128_t)rc * rc * d) return false;
        return true;
    };

    LL ans = 0, res = 0;
    for (int i = 1, j = 2; i <= n; i ++)
    {
        while (true)
        {
            int nj = j % n + 1;
            if (!check(i, nj)) break;
            LL area = Cross(p[j] - p[i], p[nj] - p[i]);
            //cout << "###" << area << ' ';
            res += abs(area);
            //cout << res << '\n';
            j = nj;
        }
        ans = max(ans, res);
        int ni = i % n + 1;
        LL parea = Cross(p[j] - p[i], p[ni] - p[i]);
        res -= abs(parea);
    }

    cout << ans << '\n';
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int T = 1; cin >> T;
    
    while (T --) solve();

    return 0;
}
```
