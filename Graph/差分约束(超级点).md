差分约束

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
vector<pair<ll, ll>> road[5010];
vector<ll> d(5010, 1e18);
ll cot[5010];
queue<ll> q;
bool en[5010] = {0};

bool spfa(ll be){
	q.push(be);
	d[be] = 0;
	en[be] = 1;
	while(q.size()){
		ll x = q.front();
		q.pop();
		en[x] = 0;
		for(auto [xt, val] : road[x]){
			ll t = d[x]+val;
			if(t < d[xt]){
				d[xt] = t;
				if(!en[xt]){
					q.push(xt);
					en[xt] = 1, cot[xt]++;
					if(cot[xt] >= n+1){
						return 0;
					}
				}
			}
		}
	}
	return 1;
}

int main(){
	cin >> n >> m;
	for(ll i = 0; i < m; i++){
		ll t1, t2 , t3;
		cin >> t1 >> t2>> t3;
		road[t2].push_back({t1, t3});
	}
	for(ll i = 1; i <= n; i++){
		road[0].push_back({i, 0});
	}
	if(!spfa(0)){
		cout << "NO" << endl;
		return 0;
	}
	for(ll i = 1; i <= n; i++){
		cout << d[i] << " ";
	}
	return 0; 
}
```
