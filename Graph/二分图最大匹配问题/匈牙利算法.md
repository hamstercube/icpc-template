#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m, ans = 0;
vector<pair<ll, ll>> v(100100);
map<ll, bool> vis;
map<ll, ll> mp;

bool dfs(ll x){ // 点与线的匹配覆盖
	auto [l, r] = v[x];
	for(ll i = l; i <= r; i++){
		if(vis[mp[i]]) continue;
		vis[mp[i]] = 1;
		if(!mp[i] || dfs(mp[i])){
			mp[i] = x;
			return true;
		}
	}
	return false;
}

void solve(){
	mp.clear();
	ans = 0;
	cin >> n;
	for(i = 1; i <= n; i++){
		cin >> v[i].first >> v[i].second;
	}
	for(i = 1; i <= n; i++){
		vis.clear();
		if(dfs(i)){
			ans++;
		}
	}
	cout << ans << endl;
}

int main(){
	ios::sync_with_stdio(0);
	cin.tie(0), cout.tie(0);
	ll T;
	cin >> T;
	while(T--){
		solve();
	}
	return 0;
}
