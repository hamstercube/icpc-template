/*
cur：当前节点
last：上一个字母代表的节点
fa ：就是 link 
nxt[c]：后缀自动机上的边，同 t r i e trietrie 树
len：endposu 中最长的字符串的长度
*/
ll i, j, n, m;
vector<vector<ll>> road;
ll ans[300100];

struct SAM{
	int fa,len;
	vector<int> nxt; //字典树建边
	SAM(){
		fa = 0; len = 0;
		nxt.assign(27,0);
	}
};

vector<SAM> sam;
vector<ll> dp;//这里的dp求的是 endpos 集合的 size
int last;
void insert(char c){
	int ch = c-'0';
	sam.push_back(SAM());
	int cur = sam.size()-1,p;
	sam[cur].len = sam[last].len+1;
	dp.push_back(1);
	for(p = last; p && sam[p].nxt[ch] == 0; p = sam[p].fa){
		sam[p].nxt[ch]=cur;
	}
	int q = sam[p].nxt[ch];
	if(q == 0){
		sam[cur].fa = 1;
	}
	else if(sam[q].len == sam[p].len+1){
		sam[cur].fa = q;
	}
	else{
		sam.push_back(SAM());
		int r = sam.size()-1;
		dp.push_back(0);
		sam[r] = sam[q];
		sam[r].len = sam[p].len+1;
		for(; p && sam[p].nxt[ch] == q; p = sam[p].fa)
			sam[p].nxt[ch] = r;
		sam[cur].fa = sam[q].fa = r;
	}
	last = cur;
}

void dfs(ll x){ //计算集合个数即为目前子串个数
	for(auto xt : road[x]){
        dfs(xt);
        dp[x] += dp[xt];
    }
    ans[sam[x].len] = max(ans[sam[x].len], dp[x]);
}
