#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

ll i, j, n, m;
map<string, string> mp;
map<string, ll> cmp;
string ss;

int main(){
	cmp["+"] = 1, cmp["-"] = 1, cmp["*"] = 2, cmp["/"] = 2, cmp["("] = 0;
	cin >> n;
	for(ll i = 0; i < n; i++){
		string s;
		ll t;
		cin >> s >> t;
		mp[s] = to_string(t);
	}
	cin >> ss;
	queue<string> q;
	stack<string> st;
	for(ll i = 0; i < ss.size(); i++){
		if(ss[i] == '{' || ss[i] == '}') continue;
		if(isdigit(ss[i])){
			string s;
			while(isdigit(ss[i])) s.push_back(ss[i++]);
			i--;
			q.push(s);
		}
		else if(isalpha(ss[i])){
			string s;
			while(isalpha(ss[i])) s.push_back(ss[i++]);
			i--;
			q.push(mp[s]);
		}
		else if(ss[i] == '('){
			st.push("(");
		}
		else if(ss[i] == ')'){
			while(st.top() != "("){
				q.push(st.top());
				st.pop();
			}
			st.pop();
		}
		else{
			string s = ss.substr(i, 1);
			//cout << s << endl;
			while(st.size() && cmp[st.top()] >= cmp[s]){
				//cout << cmp[st.top()] << " " << cmp[s] << endl;
				q.push(st.top());
				st.pop();
			}
			st.push(s);
		}
	}
	while(st.size()){
		q.push(st.top());
		st.pop();
	}
	stack<double> st1;
	while(q.size()){
		string s = q.front();
		q.pop();
		if(isdigit(s[0])){
			double d = stof(s);
			st1.push(d);
		}
		else{
			double t1, t2;
			if(st1.size()){
				t1 = st1.top();
				st1.pop();
			}
			if(st1.size()){
			 	t2 = st1.top();
			 	st1.pop();
			}
			if(s[0] == '+') t2 += t1;
			if(s[0] == '-') t2 -= t1;
			if(s[0] == '*') t2 *= t1;
			if(s[0] == '/'){
				if(t1 == 0){
					cout << "TRADE_FAILED!";
					return 0;
				}
				t2 /= t1;
			}
			st1.push(t2);
		}
	}
	printf("%.2f", st1.top());
	return 0;
}
